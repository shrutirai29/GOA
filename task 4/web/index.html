Production GSQL Query (`device_neighbors`)

CREATE QUERY device_neighbors(VERTEX<DeviceProfile> target_device) 
    FOR GRAPH FraudInvestigatorGraph RETURNS (SetAccum<VERTEX<Card>>) {
    
    SetAccum<VERTEX<Card>> @@connected_cards;
    Start = { target_device };
    
    // 1-hop: Transactions using this device profile
    Txns = SELECT t FROM Start:s -(USED_IN_TXN:e)-> Transaction:t;

    // 2-hop: Other cards that charged on this device
    Cards = SELECT c FROM Txns:t -(CHARGED_ON:e)-> Card:c
            ACCUM @@connected_cards += c;

    PRINT @@connected_cards;
}
